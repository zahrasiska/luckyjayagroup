# Kontak

Kontak terdiri dari tabel `ktk`, `ktkalamat`, `ktkkontak`, dan `ktkinfo`.

**Fitur yang didukung:**
- Multiple alamat dan multiple kontak.
- Default alamat dan default kontak.
- Relasi ke tabel devisi (tabel ktk itu sendiri).
- Relasi ke tabel wilayah (provinsi, kabupaten, kecamatan, desa).
- Partial update (patch): update hanya field yang diubah/dikirim ke tabel bersangkutan.
- Soft delete (`deleted_at`).

## Daftar Tabel

### **`ktk`**: Data master kontak (customer, supplier, karyawan, dll)

```sql
CREATE TABLE ktk (
  id          bigint NOT NULL PRIMARY KEY,
  kode        varchar(20) NOT NULL,
  nama        varchar(30),
  c           boolean,
  s           boolean,
  p           boolean,
  golongan    varchar(20),
  jabatan     varchar(20),
  perusahaan  varchar(30),
  defalamat   bigint,
  defkontak   bigint,
  aktif       boolean,
  created_at  timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at  timestamp without time zone,
  deleted_at  timestamp without time zone,
  /* Keys */
  CONSTRAINT ktk_pkey PRIMARY KEY (id)
);

CREATE INDEX ktk_fk_defalamat ON ktk (defalamat);
CREATE INDEX ktk_fk_defkontak ON ktk (defkontak);
CREATE INDEX ktk_idx_jabatan ON ktk (jabatan);
CREATE UNIQUE INDEX ktk_idx_kode ON ktk (kode);
```

```sql
CREATE INDEX ktk_fk_idkabupaten ON ktkalamat (idkabupaten);
```

### **`ktkkontak`**: Data kontak person (mendukung multiple kontak per kontak)

```sql
CREATE TABLE ktkkontak (

  idkontak    bigint NOT NULL,

  wa          varchar(20),
```

### **`ktkinfo`**: Data info tambahan kontak (devisi, hutang, piutang, saldo, plafon, tipe harga, termin, gambar)

```sql
CREATE TABLE ktkinfo (

  CONSTRAINT ktkinfo_pkey PRIMARY KEY (id),
```

## Query: Detail Kontak

```sql
SELECT 
  ktk.id,
  ktk.c, ktk.s, ktk.p,
  ktk.kode, ktk.nama, ktk.golongan, ktk.jabatan, ktk.perusahaan,
  ktk.aktif, ktk.defalamat,
  a.alamat, a.kota,
  a.idprovinsi, p.provinsi,
  a.idkabupaten, k.kabupaten, 
  a.idkecamatan, c.kecamatan,
  a.iddesa, d.desa,
  a.tipealamat, a.kodepos,
  ktk.defkontak,
  t.tipekontak, t.uraian,
  t.email, t.wa, t.telp, t.hp, t.fax, t.website,
  i.iddevisi, 
  jsonb_build_object(
    'id', dv.id,
    'kode', dv.kode,
    'nama', dv.nama
  ) AS datadevisi, -- (sales of customer)
  i.prosentase_sales, i.hutang, i.piutang, i.saldo_awal, i.saldo, i.plafon,
  i.tipe, 
  i.termin, 
  jsonb_build_object(
    'kode', tm.kode,
    'nama', tm.nama,
    'hari', tm.hari
  ) as datatermin,
  i.gambar,
  ktk.created_at, ktk.updated_at, ktk.deleted_at,
  TRIM(BOTH FROM concat_ws(
    E'\r\n',
    a.alamat,
    COALESCE(concat(c.kecamatan, ' - ', k.kabupaten), a.kota),
    TRIM(BOTH FROM COALESCE(t.uraian, ''))
  )) AS alamat_kirim,
  TRIM(BOTH FROM COALESCE(concat(c.kecamatan, ' - ', k.kabupaten), a.kota)) AS wilayah,
  -- === JSON ListAlamat ===
  COALESCE(lsalamat.json_listalamat, '[]'::jsonb) AS listalamat,          
  COALESCE(lskontak.json_listkontak, '[]'::jsonb) AS listkontak
FROM ktk ktk
LEFT JOIN ktkalamat a ON a.idkontak = ktk.id AND a.id = ktk.defalamat
LEFT JOIN prive.wilayah_provinsi p ON p.id = a.idprovinsi
LEFT JOIN prive.wilayah_kabupaten k ON k.id = a.idkabupaten
LEFT JOIN prive.wilayah_kecamatan c ON c.id = a.idkecamatan
LEFT JOIN prive.wilayah_desa d ON d.id = a.iddesa
LEFT JOIN ktkkontak t ON t.idkontak = ktk.id AND t.id = ktk.defkontak
LEFT JOIN ktkinfo i ON i.id = ktk.id
LEFT JOIN ktk dv ON dv.id = i.iddevisi
LEFT JOIN termin tm on tm.kode = i.termin
-- === JSON List alamat ===
LEFT JOIN LATERAL (
  SELECT jsonb_agg(jsonb_build_object(
    'id', k2.id,
    'tipealamat', k2.tipealamat,
    'alamat', k2.alamat,
    'idprovinsi', k2.idprovinsi,
    'provinsi', p2.provinsi,
    'idkabupaten', k2.idkabupaten,
    'kabupaten', k2kab.kabupaten,
    'idkecamatan', k2.idkecamatan,
    'kecamatan', c2.kecamatan,
    'iddesa', k2.iddesa,
    'desa', d2.desa,
    'kodepos', k2.kodepos
  )) AS json_listalamat
  FROM ktkalamat k2
    LEFT JOIN prive.wilayah_provinsi p2 ON p2.id = k2.idprovinsi
    LEFT JOIN prive.wilayah_kabupaten k2kab ON k2kab.id = k2.idkabupaten
    LEFT JOIN prive.wilayah_kecamatan c2 ON c2.id = k2.idkecamatan
    LEFT JOIN prive.wilayah_desa d2 ON d2.id = k2.iddesa
  WHERE k2.idkontak = ktk.id AND k2.deleted_at IS NULL
) lsalamat ON TRUE
-- === JSON List kontak ===
LEFT JOIN LATERAL (
  SELECT jsonb_agg(jsonb_build_object(
    'id', k3.id,
    'tipekontak', k3.tipekontak,
    'uraian', k3.uraian,
    'email', k3.email,
    'wa', k3.wa,
    'telp', k3.telp,
    'hp', k3.hp,
    'fax', k3.fax,
    'website', k3.website
  )) AS json_listkontak
  FROM ktkkontak k3            
  WHERE k3.idkontak = ktk.id AND k3.deleted_at IS NULL
) lskontak ON TRUE
WHERE ktk.id = $1 AND ktk.deleted_at IS NULL
LIMIT 1;
```
